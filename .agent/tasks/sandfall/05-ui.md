### `src/sandfall/ui.py`

The palette is a horizontal bar of swatches, one per non-EMPTY element, in `ElementId` order. Click a swatch to select. The active swatch gets a highlight border.

To keep this **headless-testable**, separate the *layout* (pure: given window width & element list, return `list[(ElementId, pygame.Rect-like)]`) from the *drawing* (needs pygame). Test the layout function; visually verify drawing.

```python
"""In-game UI: element palette + FPS overlay."""

from __future__ import annotations

from dataclasses import dataclass

from .config import (
    FONT_NAME, FONT_SIZE, FPS_COLOR, PALETTE_MARGIN, PALETTE_PADDING,
    PALETTE_SWATCH,
)
from .elements import ELEMENTS, ElementId


@dataclass(frozen=True, slots=True)
class Swatch:
    element_id: ElementId
    x: int
    y: int
    w: int
    h: int

    def contains(self, px: int, py: int) -> bool:
        return self.x <= px <= self.x + self.w and self.y <= py <= self.h_y_end()

    def h_y_end(self) -> int:
        return self.y + self.h


def palette_layout(window_width: int, bar_y: int) -> list[Swatch]:
    """Pure: compute swatch rects for the non-EMPTY elements, left-to-right."""
    swatches: list[Swatch] = []
    x = PALETTE_MARGIN
    for eid in ElementId:
        if eid == ElementId.EMPTY:
            continue
        swatches.append(Swatch(eid, x, bar_y, PALETTE_SWATCH, PALETTE_SWATCH))
        x += PALETTE_SWATCH + PALETTE_PADDING
    return swatches


class UI:
    """Owns the palette highlight + FPS text rendering."""

    def __init__(self, window_width: int, window_height: int) -> None:
        self._window_width = window_width
        self._bar_y = window_height - (PALETTE_SWATCH + 2 * PALETTE_MARGIN)
        self._swatches = palette_layout(window_width, self._bar_y)
        # Font is created lazily in draw() so pygame must be init first.
        self._font: object | None = None

    @property
    def swatches(self) -> list[Swatch]:
        return self._swatches

    def swatch_at(self, px: int, py: int) -> ElementId | None:
        for s in self._swatches:
            if s.contains(px, py):
                return s.element_id
        return None

    def draw(self, screen, active: ElementId, fps: float) -> None:
        import pygame
        if self._font is None:
            self._font = pygame.font.Font(FONT_NAME, FONT_SIZE)
        # FPS overlay
        assert self._font is not None
        fps_surf = self._font.render(f"{int(fps)} FPS", True, FPS_COLOR)
        screen.blit(fps_surf, (PALETTE_MARGIN, PALETTE_MARGIN))
        # Palette swatches
        for s in self._swatches:
            color = ELEMENTS[s.element_id].color
            pygame.draw.rect(screen, color, (s.x, s.y, s.w, s.h))
            if s.element_id == active:
                pygame.draw.rect(screen, (255, 255, 255), (s.x, s.y, s.w, s.h), 2)
```

> Note: `contains`/`h_y_end` in the snippet above are slightly awkward; simplify to a clean `x0 <= px <= x1 and y0 <= py <= y1` using stored `x1 = x + w`, `y1 = y + h` fields. The implementer should make `Swatch` clean and correct — the key requirement is that `palette_layout` and `swatch_at` are pure and unit-testable without pygame.

### `src/sandfall/game.py` — EDIT

In `Game.__init__`: create `self._ui = UI(WINDOW_WIDTH, WINDOW_HEIGHT)`; add `self._paused = False`.

In `_handle_events`, add:

```python
elif event.type == pygame.MOUSEWHEEL:
    delta = event.y  # +1 up, -1 down (pygame-ce)
    self._brush_radius = clamp(self._brush_radius + delta, BRUSH_MIN, BRUSH_MAX)
elif event.type == pygame.MOUSEBUTTONDOWN and event.button == 1:
    # palette click takes priority over painting
    sel = self._ui.swatch_at(*event.pos)
    if sel is not None:
        self._selected = sel
elif event.type == pygame.KEYDOWN:
    if event.key == pygame.K_SPACE:
        self._paused = not self._paused
    elif event.key == pygame.K_n and self._paused:
        self._sim.step()
```

Where `clamp(v, lo, hi) = max(lo, min(hi, v))` (define inline or in `config.py`).

In the main loop:
- Replace unconditional `self._sim.step()` with:
  ```python
  if not self._paused:
      self._sim.step()
  ```
- After blitting the grid, call `self._ui.draw(self._screen, self._selected, self._clock.get_fps())`.
- Adjust painting so a palette click does NOT also paint: the cleanest approach is to **not** paint on the same frame a palette click happened. Track `self._just_clicked_palette` flag set in the event handler and skip paint that frame. (Or: only paint on `MOUSEBUTTONDOWN` outside the palette + while dragging. Pick a clean approach; document it.)

Display the brush radius somewhere small (e.g. next to FPS: `{fps} FPS  r={radius}`) so the scroll-wheel effect is visible.
